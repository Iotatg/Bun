# Bun Language Phase 1.5 — Strings, Dicts, File I/O

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Grow Bun from a toy interpreter to a language that can do real text-processing work, driven by converting tools from the oven-sh/bun repository. Add string methods, list methods, a dict type, and file I/O.

**Architecture:** Method-call syntax (`receiver.method(args)`) via a new `DOT` token and a `MethodCall` AST node. The interpreter dispatches on receiver type (str/list/dict) to a method table. A `BunDict` value type in `values.py` wraps a Python dict. File I/O is two builtins (`read_file`, `write_file`). No new dependencies.

**Tech Stack:** Python 3.8+, pytest.

## Global Constraints

- Python >=3.8; standard library only. Zero third-party deps.
- Every runtime error raises `BunRuntimeError` with line/col (no raw Python tracebacks leak).
- Method calls use `receiver.method(args)` syntax. No free-function string helpers.
- Unknown method on a value raises `BunRuntimeError` naming the value type.
- Dict keys may be strings or numbers; ints and strings are distinct keys.
- Dict literal syntax: `{}` and `{"key": value}`. Parsed in expression position only.
- No code comments unless the plan's code includes them.
- Use `python3` (no `python` on PATH).

```python
from .errors import BunRuntimeError


def _arity(name, args, n):
    if len(args) != n:
        raise BunRuntimeError(
            "'%s' expects %d argument(s), got %d" % (name, n, len(args)))


def _check_str(name, args, i):
    if not isinstance(args[i], str):
        raise BunRuntimeError("'%s' expects a string argument" % name)


def _check_int(name, args, i):
    if not isinstance(args[i], int) or isinstance(args[i], bool):
        raise BunRuntimeError("'%s' expects an integer argument" % name)


def _check_list(name, receiver):
    if not isinstance(receiver, list):
        raise BunRuntimeError("'%s' is only available on lists" % name)

    return recv.strip()

    return recv.lower()

    _arity("upper", args, 0)
    return recv.upper()
```

```python
    def _method_call(self, node, env):
        inner = node.callee
        receiver = self.evaluate(inner.receiver, env)
        args = [self.evaluate(arg, env) for arg in inner.args]
        if isinstance(receiver, str):
            table = STRING_METHODS
            type_name = "string"
        elif isinstance(receiver, list):
            table = LIST_METHODS
            type_name = "list"
        else:
            raise BunRuntimeError(
                "'%s' is not a method on %s" % (inner.method, self._type_name(receiver)))
        if inner.method not in table:
            raise BunRuntimeError(
                "unknown method '%s' on %s" % (inner.method, type_name))
        try:
            return table[inner.method](receiver, args)
        except BunRuntimeError as error:
            if error.line == 0:
                error.line, error.col = inner.line, inner.col
            raise

    def _type_name(self, value):
        if isinstance(value, bool):
            return "bool"
        if isinstance(value, int):
            return "int"
        if isinstance(value, float):
            return "float"
        if isinstance(value, str):
            return "string"
        if value is None:
            return "null"
        if isinstance(value, list):
            return "list"
        return type(value).__name__
```

```python
from .errors import BunRuntimeError
from .methods import _arity, _check_str, _check_int


def make_key(key):
    if isinstance(key, bool):
        return ("b", key)
    if isinstance(key, (int, float)):
        return ("n", key)
    if isinstance(key, str):
        return ("s", key)
    if key is None:
        return ("z", None)
    raise BunRuntimeError("dict keys must be strings, numbers, or null")


class BunDict:
    def __init__(self):
        self.items = {}

    def get(self, key):
        return self.items[make_key(key)]

    def set(self, key, value):
        self.items[make_key(key)] = value

    def has(self, key):
        return make_key(key) in self.items

    def remove(self, key):
        k = make_key(key)
        if k not in self.items:
            raise BunRuntimeError("key not found")
        del self.items[k]

    def keys(self):
        return [k[1] for k in self.items]

    def values(self):
        return list(self.items.values())

    def __len__(self):
        return len(self.items)


def _check_dict(name, receiver):
    if not isinstance(receiver, BunDict):
        raise BunRuntimeError("'%s' is only available on dicts" % name)


def d_len(recv, args):
    _arity("len", args, 0)
    return len(recv)


def d_has(recv, args):
    _arity("has", args, 1)
    return recv.has(args[0])


def d_get(recv, args):
    _arity("get", args, 1)
    try:
        return recv.get(args[0])
    except KeyError:
        raise BunRuntimeError("key not found")


def d_keys(recv, args):
    _arity("keys", args, 0)
    return recv.keys()


def d_values(recv, args):
    _arity("values", args, 0)
    return recv.values()


def d_remove(recv, args):
    _arity("remove", args, 1)
    try:
        recv.remove(args[0])
    except KeyError:
        raise BunRuntimeError("key not found")
    return None


DICT_METHODS = {
    "len": d_len,
    "has": d_has,
    "get": d_get,
    "keys": d_keys,
    "values": d_values,
    "remove": d_remove,
}
```

```python
            handle.write(args[1])
    except OSError as error:
        raise BunRuntimeError("cannot write '%s': %s" % (args[0], error.strerror))
    return None


def _append_file(args):
    _two("append_file", args)
    try:
        with open(args[0], "a") as handle:
            handle.write(args[1])
    except OSError as error:
        raise BunRuntimeError("cannot append to '%s': %s" % (args[0], error.strerror))
    return None


def _file_exists(args):
    _one("file_exists", args)
    return os.path.exists(args[0])


def _one(name, args):
    if len(args) != 1:
        raise BunRuntimeError("%s() expects 1 argument, got %d" % (name, len(args)))


def _two(name, args):
    if len(args) != 2:
        raise BunRuntimeError("%s() expects 2 arguments, got %d" % (name, len(args)))
```

```python
@pytest.mark.parametrize("path", golden_cases(), ids=os.path.basename)

    run_source(source, filename=path)
    assert capsys.readouterr().out == expected
```
