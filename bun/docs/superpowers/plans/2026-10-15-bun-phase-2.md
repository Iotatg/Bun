# Bun Phase 2 — Gradual Static Type Checker Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Add a gradual static type checker: optional Rust-like annotations (`let x: int`, `fn f(a: int) -> int`), unannotated code is `any`, checker runs before the interpreter by default and blocks type errors, `--no-check` bypasses.

**Architecture:** New `bun/types.py` holds the type universe, the assignability rule, and builtin signatures. New `bun/checker.py` walks the AST with copy-on-branch scopes mirroring the interpreter's lexical scoping, recording declared types and known function signatures, and raising `BunTypeError` on mismatches. The CLI runs lex -> parse -> check -> run. No interpreter changes: annotations live in AST fields the interpreter ignores.

**Tech Stack:** Python 3.8+, pytest, standard library only.

## Global Constraints

- Python >=3.8; stdlib only; zero third-party deps.
- All checker errors raise `BunTypeError` (BunError family) with line/col; no raw Python exceptions escape.
- Unannotated names/params are `any`; `any` is assignable both directions.
- `int` is assignable to `float`; `float` is not assignable to `int`.
- `null` is assignable to every type (nullable by default).
- No code comments unless the plan's code includes them.
- Use `python3` (no `python` on PATH).
- The existing test suite must stay green at every step (unannotated code is all `any`, so nothing that passes today may newly fail).

```python
INT = "int"
FLOAT = "float"
STRING = "string"
BOOL = "bool"
NULL = "null"
ANY = "any"
LIST = "list"
DICT = "dict"
FN = "fn"

TYPE_NAMES = frozenset({INT, FLOAT, STRING, BOOL, NULL, ANY, LIST, DICT, FN})

BUILTIN_SIGNATURES = {
    "print": [((ANY,), NULL)],
    "len": [((ANY,), INT)],
    "range": [((INT,), LIST), ((INT, INT), LIST)],
    "str": [((ANY,), STRING)],
    "int": [((ANY,), INT)],
    "float": [((ANY,), FLOAT)],
    "read_file": [((STRING,), STRING)],
    "write_file": [((STRING, STRING), NULL)],
    "append_file": [((STRING, STRING), NULL)],
    "file_exists": [((STRING,), BOOL)],
}


def is_known_type(name):
    return name in TYPE_NAMES


def is_assignable(source, target):
    if source == target:
        return True
    if source == ANY or target == ANY:
        return True
    if source == INT and target == FLOAT:
        return True
    if source == NULL:
        return True
    return False
```

- [ ] **Step 4: Write bun/checker.py**

```python
from dataclasses import dataclass
from typing import List, Optional

from . import ast, types
from .errors import BunTypeError


@dataclass
class Binding:
    type: str
    param_types: Optional[List[Optional[str]]] = None
    return_type: Optional[str] = None


class Checker:
    def __init__(self):
        self.base_dir = None
        self.return_stack = []
        self.module_cache = {}
        self.module_loading = set()

    def check(self, program, base_dir=None):
        self.base_dir = base_dir
        scope = {}
        self._collect_functions(program.statements, scope)
        self._statements(program.statements, scope)
        return scope

    def _collect_functions(self, statements, scope):
        for statement in statements:
            if isinstance(statement, ast.FnDecl):
                scope[statement.name] = Binding(
                    types.FN, self._param_types(statement), statement.return_type)

    def _param_types(self, fn):
        if fn.param_types is None:
            return [None] * len(fn.params)
        return fn.param_types

    def _statements(self, statements, scope):
        for statement in statements:
            self._statement(statement, scope)

    def _statement(self, node, scope):
        if isinstance(node, ast.LetStmt):
            self._check_annotation(node.type_ann, node)
            value_type = self._expr(node.value, scope)
            if (node.type_ann is not None
                    and not types.is_assignable(value_type, node.type_ann)):
                raise BunTypeError(
                    "cannot assign value of type '%s' to variable '%s' of type '%s'"
                    % (value_type, node.name, node.type_ann), node.line, node.col)
            scope[node.name] = Binding(node.type_ann or types.ANY)
        elif isinstance(node, ast.FnDecl):
            self._check_annotation(node.return_type, node)
            self._function(node, scope)
        elif isinstance(node, ast.AssignStmt):
            value_type = self._expr(node.value, scope)
            if isinstance(node.target, ast.Identifier):
                binding = scope.get(node.target.name)
                if (binding is not None and binding.type != types.ANY
                        and not types.is_assignable(value_type, binding.type)):
                    raise BunTypeError(
                        "cannot assign value of type '%s' to variable '%s' of type '%s'"
                        % (value_type, node.target.name, binding.type),
                        node.line, node.col)
            else:
                self._expr(node.target, scope)
        elif isinstance(node, ast.ReturnStmt):
            if node.value is not None:
                value_type = self._expr(node.value, scope)
            else:
                value_type = types.NULL
            if self.return_stack:
                declared = self.return_stack[-1]
                if (declared is not None
                        and not types.is_assignable(value_type, declared)):
                    raise BunTypeError(
                        "cannot return value of type '%s' from function with return type '%s'"
                        % (value_type, declared), node.line, node.col)
        elif isinstance(node, ast.Block):
            self._statements(node.statements, dict(scope))
        elif isinstance(node, ast.IfStmt):
            self._expr(node.condition, scope)
            self._statement(node.then_branch, dict(scope))
            if node.else_branch is not None:
                self._statement(node.else_branch, dict(scope))
        elif isinstance(node, ast.WhileStmt):
            self._expr(node.condition, scope)
            self._statements(node.body.statements, dict(scope))
        elif isinstance(node, ast.ForStmt):
            self._expr(node.iterable, scope)
            self._statements(node.body.statements, dict(scope))
        elif isinstance(node, ast.ExprStmt):
            self._expr(node.expr, scope)
        elif isinstance(node, ast.Program):
            self._statements(node.statements, dict(scope))
        else:
            raise BunTypeError(
                "cannot check statement: %s" % type(node).__name__,
                getattr(node, "line", 0), getattr(node, "col", 0))

    def _function(self, fn, scope):
        local = dict(scope)
        for name, ptype in zip(fn.params, self._param_types(fn)):
            self._check_annotation(ptype, fn)
            local[name] = Binding(ptype or types.ANY)
        self.return_stack.append(fn.return_type)
        try:
            self._statements(fn.body.statements, local)
        finally:
            self.return_stack.pop()

    def _check_annotation(self, name, node):
        if name is not None and not types.is_known_type(name):
            raise BunTypeError("unknown type '%s'" % name, node.line, node.col)

    def _expr(self, node, scope):
        if isinstance(node, ast.Literal):
            return self._literal_type(node.value)
        if isinstance(node, ast.Identifier):
            binding = scope.get(node.name)
            return binding.type if binding is not None else types.ANY
        if isinstance(node, ast.ListLiteral):
            for element in node.elements:
                self._expr(element, scope)
            return types.LIST
        if isinstance(node, ast.DictLiteral):
            for key, value in node.entries:
                self._expr(key, scope)
                self._expr(value, scope)
            return types.DICT
        if isinstance(node, ast.FnDecl):
            for ptype in self._param_types(node):
                self._check_annotation(ptype, node)
            self._check_annotation(node.return_type, node)
            self._function(node, scope)
            return types.FN
        if isinstance(node, ast.Call):
            return self._call(node, scope)
        if isinstance(node, ast.Unary):
            self._expr(node.operand, scope)
            return types.ANY
        if isinstance(node, ast.Binary):
            self._expr(node.left, scope)
            self._expr(node.right, scope)
            return types.ANY
        if isinstance(node, ast.Logical):
            self._expr(node.left, scope)
            self._expr(node.right, scope)
            return types.ANY
        if isinstance(node, ast.Index):
            self._expr(node.target, scope)
            self._expr(node.index, scope)
            return types.ANY
        if isinstance(node, ast.MethodCall):
            self._expr(node.receiver, scope)
            for arg in node.args:
                self._expr(arg, scope)
            return types.ANY
        raise BunTypeError(
            "cannot check expression: %s" % type(node).__name__,
            getattr(node, "line", 0), getattr(node, "col", 0))

    def _literal_type(self, value):
        if value is None:
            return types.NULL
        if isinstance(value, bool):
            return types.BOOL
        if isinstance(value, int):
            return types.INT
        if isinstance(value, float):
            return types.FLOAT
        if isinstance(value, str):
            return types.STRING
        return types.ANY

    def _call(self, node, scope):
        arg_types = [self._expr(arg, scope) for arg in node.args]
        if not isinstance(node.callee, ast.Identifier):
            self._expr(node.callee, scope)
            return types.ANY
        name = node.callee.name
        if name in types.BUILTIN_SIGNATURES:
            return self._builtin_call(name, node, arg_types)
        binding = scope.get(name)
        if binding is None:
            return types.ANY
        if binding.type != types.FN:
            raise BunTypeError(
                "'%s' of type '%s' is not callable" % (name, binding.type),
                node.line, node.col)
        if binding.param_types is None:
            return types.ANY
        if len(arg_types) != len(binding.param_types):
            raise BunTypeError(
                "function '%s' expects %d argument(s), got %d"
                % (name, len(binding.param_types), len(arg_types)),
                node.line, node.col)
        for arg_type, ptype in zip(arg_types, binding.param_types):
            if ptype is None:
                continue
            if not types.is_assignable(arg_type, ptype):
                raise BunTypeError(
                    "argument of type '%s' is not assignable to parameter of type '%s' in call to '%s'"
                    % (arg_type, ptype, name), node.line, node.col)
        return binding.return_type or types.ANY

    def _builtin_call(self, name, node, arg_types):
        for params, return_type in types.BUILTIN_SIGNATURES[name]:
            if len(params) != len(arg_types):
                continue
            for arg_type, ptype in zip(arg_types, params):
                if not types.is_assignable(arg_type, ptype):
                    break
            else:
                return return_type
        expected = " or ".join(
            str(len(params)) for params, _ in types.BUILTIN_SIGNATURES[name])
        raise BunTypeError(
            "%s() received arguments that do not match its signature (expected %s argument(s))"
            % (name, expected), node.line, node.col)
```

```python
import os
import sys

from .lexer import Lexer
from .parser import Parser
from .checker import Checker
from .interpreter import Interpreter
from .values import stringify, ReturnSignal
from .errors import BunError, BunRuntimeError, format_error
from . import ast

USAGE = """Bun programming language

Usage:
  bun <file.bun>              run a Bun program
  bun --no-check <file.bun>   run without type checking
  bun                         start the interactive REPL
  bun -h | --help             show this help
"""


def run_source(source, filename="<stdin>", check=True):
    tokens = Lexer(source).tokenize()
    program = Parser(tokens).parse()
    base_dir = os.path.dirname(os.path.abspath(filename))
    if check:
        Checker().check(program, base_dir=base_dir)
    interpreter = Interpreter()
    interpreter.run(program, base_dir=base_dir)
    return interpreter.globals


def run_file(path, check=True):
    try:
        with open(path, "r") as handle:
            source = handle.read()
    except OSError as error:
        print("error: cannot open file '%s': %s" % (path, error), file=sys.stderr)
        return 1
    try:
        run_source(source, filename=path, check=check)
        return 0
    except BunError as error:
        print(format_error(error, source, path), file=sys.stderr)
        return 1


def repl(check=True):
    interpreter = Interpreter()
    while True:
        try:
            line = input("bun> ")
        except EOFError:
            print()
            break
        if line.strip() == "":
            continue
        try:
            program = Parser(Lexer(line).tokenize()).parse()
            if check:
                Checker().check(program, base_dir=os.getcwd())
            for statement in program.statements:
                if isinstance(statement, ast.ExprStmt):
                    value = interpreter.evaluate(statement.expr, interpreter.globals)
                    if value is not None:
                        print(stringify(value))
                else:
                    try:
                        interpreter.execute(statement, interpreter.globals)
                    except ReturnSignal:
                        raise BunRuntimeError(
                            "'return' outside of a function",
                            statement.line, statement.col)
        except BunError as error:
            print(format_error(error, line, "<repl>"))
    return 0


def main(argv=None):
    args = list(sys.argv[1:] if argv is None else argv)
    check = True
    if "--no-check" in args:
        args = [arg for arg in args if arg != "--no-check"]
        check = False
    if not args:
        return repl(check)
    if args[0] in ("-h", "--help"):
        print(USAGE)
        return 0
    return run_file(args[0], check=check)
```
