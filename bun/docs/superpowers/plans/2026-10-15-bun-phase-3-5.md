# Bun Phase 3.5 Implementation Plan: Bytecode Compiler and Stack VM

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a bytecode compiler (`bun/compiler.py`) and a name-based stack VM (`bun/vm.py`) that execute the full Bun language, becoming the default engine so Phase 4 (self-hosting) has a plain-data compile/run interface.

**Architecture:** The compiler walks the validated AST and emits a CodeObject (dict) whose instructions are 4-tuples `(op, arg, line, col)`. The VM is a flat dispatch loop over call frames, reusing the existing `Environment` value model and delegating all operator/method/index semantics to shared helpers in `bun/ops.py` (extracted from the interpreter in Task 2) so the VM cannot drift from the tree-walker. The tree-walker is retained as a differential oracle; every golden runs under both engines.

**Tech Stack:** Python >= 3.8, standard library only. pytest for tests. No third-party runtime dependencies.

## Global Constraints

- Python >= 3.8, standard library only; zero third-party runtime dependencies.
- Every language-level error is a `BunError` subclass (`BunRuntimeError` for runtime errors); no raw Python exception may escape to the user. VM-internal control flow (`_TopLevelReturn`) never escapes `VM.run`.
- Error messages and positions must match the tree-walker byte-for-byte; differential goldens enforce this.
- The tree-walker stays in the tree as the differential oracle. Do not delete it.
- No code comments in committed code.
- Execute on branch `bun-phase-3-5` (create it before Task 1).
- One commit per task; commit message style follows the repo (`feat:`, `fix:`, `chore:`, `docs:`, `test:`).
- After each task, run the full suite (`python3 -m pytest tests/ -q`) and record the pass count in the task's final step.

- [ ] **Step 3: Write `bun/compiler.py`**

```python
from . import ast


OP_LOAD_CONST = "LOAD_CONST"
OP_BUILD_LIST = "BUILD_LIST"
OP_BUILD_DICT = "BUILD_DICT"
OP_LOAD_NAME = "LOAD_NAME"
OP_DEFINE_NAME = "DEFINE_NAME"
OP_STORE_NAME = "STORE_NAME"
OP_POP = "POP"
OP_DUP = "DUP"
OP_MAKE_FUNCTION = "MAKE_FUNCTION"
OP_CALL = "CALL"
OP_RETURN = "RETURN"
OP_HALT = "HALT"
OP_ENTER_SCOPE = "ENTER_SCOPE"
OP_EXIT_SCOPE = "EXIT_SCOPE"


class CompileError(Exception):
    pass


def make_code(name, params, code, filename, source):
    return {
        "name": name,
        "params": list(params),
        "code": code,
        "filename": filename,
        "source": source,
    }


def emit(code, op, arg, node):
    code.append((op, arg, getattr(node, "line", 0), getattr(node, "col", 0)))
    return len(code) - 1


def patch(code, index):
    op, arg, line, col = code[index]
    code[index] = (op, len(code), line, col)


class Compiler:
    def __init__(self, filename="<stdin>", source=""):
        self.filename = filename
        self.source = source

    def compile_program(self, program):
        code = []
        for statement in program.statements:
            self.compile_statement(statement, code)
        code.append((OP_HALT, None, 0, 0))
        return make_code("<module>", [], code, self.filename, self.source)

    def compile_function(self, node):
        code = []
        emit(code, OP_ENTER_SCOPE, None, node)
        body = node.body
        if isinstance(body, ast.Block):
            for statement in body.statements:
                self.compile_statement(statement, code)
        else:
            self.compile_statement(body, code)
        emit(code, OP_EXIT_SCOPE, None, node)
        emit(code, OP_LOAD_CONST, None, node)
        emit(code, OP_RETURN, None, node)
        return make_code(node.name or "anonymous", node.params, code,
                         self.filename, self.source)

    def compile_statement(self, node, code):
        if isinstance(node, ast.LetStmt):
            self.compile_expr(node.value, code)
            emit(code, OP_DEFINE_NAME, (node.name, node.mutable), node)
        elif isinstance(node, ast.AssignStmt):
            if isinstance(node.target, ast.Identifier):
                self.compile_expr(node.value, code)
                emit(code, OP_STORE_NAME, node.target.name, node)
            else:
                raise CompileError("invalid assignment target")
        elif isinstance(node, ast.ExprStmt):
            self.compile_expr(node.expr, code)
            emit(code, OP_POP, None, node)
        elif isinstance(node, ast.Block):
            emit(code, OP_ENTER_SCOPE, None, node)
            for statement in node.statements:
                self.compile_statement(statement, code)
            emit(code, OP_EXIT_SCOPE, None, node)
        elif isinstance(node, ast.FnDecl):
            emit(code, OP_MAKE_FUNCTION, self.compile_function(node), node)
            emit(code, OP_DEFINE_NAME, (node.name, False), node)
        elif isinstance(node, ast.ReturnStmt):
            if node.value is not None:
                self.compile_expr(node.value, code)
            else:
                emit(code, OP_LOAD_CONST, None, node)
            emit(code, OP_RETURN, None, node)
        else:
            raise CompileError(
                "unknown statement: %s" % type(node).__name__)

    def compile_expr(self, node, code):
        if isinstance(node, ast.Literal):
            emit(code, OP_LOAD_CONST, node.value, node)
        elif isinstance(node, ast.Identifier):
            emit(code, OP_LOAD_NAME, node.name, node)
        elif isinstance(node, ast.ListLiteral):
            for element in node.elements:
                self.compile_expr(element, code)
            emit(code, OP_BUILD_LIST, len(node.elements), node)
        elif isinstance(node, ast.DictLiteral):
            for key_node, value_node in node.entries:
                self.compile_expr(key_node, code)
                self.compile_expr(value_node, code)
            emit(code, OP_BUILD_DICT, len(node.entries), node)
        elif isinstance(node, ast.Call):
            self.compile_expr(node.callee, code)
            for arg_node in node.args:
                self.compile_expr(arg_node, code)
            emit(code, OP_CALL, len(node.args), node)
        else:
            raise CompileError(
                "unknown expression: %s" % type(node).__name__)
```

Parameters are bound by the VM's `CALL` handler in the call environment (design spec section 3), not by the compiled function body, so `compile_function` does not emit parameter definitions.

- [ ] **Step 4: Write `bun/vm.py`**

```python
import os

from .builtins import register_builtins
from .compiler import (
    OP_BUILD_DICT,
    OP_BUILD_LIST,
    OP_CALL,
    OP_DEFINE_NAME,
    OP_DUP,
    OP_ENTER_SCOPE,
    OP_EXIT_SCOPE,
    OP_HALT,
    OP_LOAD_CONST,
    OP_LOAD_NAME,
    OP_MAKE_FUNCTION,
    OP_POP,
    OP_RETURN,
    OP_STORE_NAME,
    Compiler,
)
from .dict import BunDict
from .environment import Environment
from .errors import BunRuntimeError
from .values import BunFunction, Builtin, stringify


class _TopLevelReturn(Exception):
    def __init__(self, error):
        super().__init__()
        self.error = error


_DONE = object()


class Frame:
    def __init__(self, code, env, base_dir, caller=None):
        self.code = code["code"]
        self.ip = 0
        self.env = env
        self.base_dir = base_dir
        self.stack = []
        self.handlers = []
        self.caller = caller
        self.iterators = []
        self.depth = 1 if caller is None else caller.depth + 1


class VM:
    MAX_FRAMES = 10000

    def __init__(self):
        self.globals = Environment()
        register_builtins(self.globals)
        self.globals.module_dir = os.getcwd()

    def run(self, code, env=None, base_dir=None):
        env = env if env is not None else self.globals
        if base_dir is None:
            base_dir = os.getcwd()
        self.globals.module_dir = base_dir
        return self._execute(code, env, base_dir)

    def eval_expression(self, node, env, base_dir=None):
        code = Compiler("<repl>", "").compile_expression(node)
        return self._execute(code, env, base_dir if base_dir is not None
                             else os.getcwd())

    def _execute(self, code, env, base_dir):
        frame = Frame(code, env, base_dir)
        while True:
            op, arg, line, col = frame.code[frame.ip]
            frame.ip += 1
            handler = self.DISPATCH[op]
            try:
                next_frame = handler(self, frame, arg, line, col)
            except _TopLevelReturn as signal:
                raise signal.error
            except BunRuntimeError as error:
                if error.line == 0:
                    error.line, error.col = line, col
                target = frame
                while target is not None and not target.handlers:
                    target = target.caller
                if target is None:
                    raise
                catch_ip, catch_env, stack_depth = target.handlers.pop()
                del target.stack[stack_depth:]
                target.env = catch_env
                target.ip = catch_ip
                target.stack.append(error.msg)
                frame = target
                continue
            if next_frame is _DONE:
                return frame.stack.pop() if frame.stack else None
            if next_frame is not None:
                frame = next_frame

    def _op_load_const(self, frame, arg, line, col):
        frame.stack.append(arg)

    def _op_build_list(self, frame, arg, line, col):
        if arg:
            items = frame.stack[-arg:]
            del frame.stack[-arg:]
        else:
            items = []
        frame.stack.append(items)

    def _op_build_dict(self, frame, arg, line, col):
        result = BunDict()
        if arg:
            flat = frame.stack[-arg * 2:]
            del frame.stack[-arg * 2:]
            for position in range(0, len(flat), 2):
                result.set(flat[position], flat[position + 1])
        frame.stack.append(result)

    def _op_load_name(self, frame, arg, line, col):
        frame.stack.append(frame.env.get(arg))

    def _op_define_name(self, frame, arg, line, col):
        name, mutable = arg
        frame.env.define(name, frame.stack.pop(), mutable)

    def _op_store_name(self, frame, arg, line, col):
        frame.env.assign(arg, frame.stack.pop())

    def _op_pop(self, frame, arg, line, col):
        frame.stack.pop()

    def _op_dup(self, frame, arg, line, col):
        frame.stack.append(frame.stack[-1])

    def _op_enter_scope(self, frame, arg, line, col):
        frame.env = Environment(frame.env)

    def _op_exit_scope(self, frame, arg, line, col):
        frame.env = frame.env.parent

    def _op_make_function(self, frame, arg, line, col):
        frame.stack.append(
            BunFunction(arg["name"], arg["params"], arg, frame.env))

    def _op_call(self, frame, arg, line, col):
        if arg:
            args = frame.stack[-arg:]
            del frame.stack[-arg:]
        else:
            args = []
        callee = frame.stack.pop()
        if isinstance(callee, BunFunction):
            if len(args) != len(callee.params):
                raise BunRuntimeError(
                    "function '%s' expects %d argument(s), got %d"
                    % (callee.name, len(callee.params), len(args)))
            call_env = Environment(callee.closure)
            for name, value in zip(callee.params, args):
                call_env.define(name, value, True)
            if frame.depth >= self.MAX_FRAMES:
                raise BunRuntimeError("stack overflow")
            return Frame(callee.body, call_env, frame.base_dir, caller=frame)
        if isinstance(callee, Builtin):
            frame.stack.append(callee.func(args))
            return None
        raise BunRuntimeError(
            "%s is not callable" % stringify(callee))

    def _op_return(self, frame, arg, line, col):
        value = frame.stack.pop()
        if frame.caller is None:
            raise _TopLevelReturn(
                BunRuntimeError("'return' outside of a function", line, col))
        frame.caller.stack.append(value)
        return frame.caller

    def _op_halt(self, frame, arg, line, col):
        return _DONE

    DISPATCH = {
        OP_LOAD_CONST: _op_load_const,
        OP_BUILD_LIST: _op_build_list,
        OP_BUILD_DICT: _op_build_dict,
        OP_LOAD_NAME: _op_load_name,
        OP_DEFINE_NAME: _op_define_name,
        OP_STORE_NAME: _op_store_name,
        OP_POP: _op_pop,
        OP_DUP: _op_dup,
        OP_ENTER_SCOPE: _op_enter_scope,
        OP_EXIT_SCOPE: _op_exit_scope,
        OP_MAKE_FUNCTION: _op_make_function,
        OP_CALL: _op_call,
        OP_RETURN: _op_return,
        OP_HALT: _op_halt,
    }
```

Note that `VM.DISPATCH` stores plain functions defined in the class body; they are called as `handler(self, frame, arg, line, col)` with `self` and `frame` passed explicitly, so no bound-method wrapping is needed. The top-level import is `from .values import BunFunction, Builtin, stringify`.

- [ ] **Step 1: Write `bun/ops.py` (extraction)**

```python
from .dict import BunDict, DICT_METHODS
from .errors import BunRuntimeError
from .methods import LIST_METHODS, STRING_METHODS
from .values import (
    BunFunction, is_number, is_truthy, stringify, values_equal,
)


def binary_op(op, left, right):
    if op == "+":
        if isinstance(left, str) and isinstance(right, str):
            return left + right
        if isinstance(left, list) and isinstance(right, list):
            return left + right
        if is_number(left) and is_number(right):
            return left + right
        raise BunRuntimeError("unsupported operands for '+'")
    if op in ("-", "*", "%"):
        if not (is_number(left) and is_number(right)):
            raise BunRuntimeError("unsupported operands for '%s'" % op)
        if op == "%":
            if right == 0:
                raise BunRuntimeError("division by zero")
            return left % right
        return left - right if op == "-" else left * right
    if op == "/":
        if not (is_number(left) and is_number(right)):
            raise BunRuntimeError("unsupported operands for '/'")
        if right == 0:
            raise BunRuntimeError("division by zero")
        if isinstance(left, int) and isinstance(right, int):
            return left // right
        return left / right
    if op in ("<", "<=", ">", ">="):
        if (is_number(left) and is_number(right)) or (
                isinstance(left, str) and isinstance(right, str)):
            if op == "<":
                return left < right
            if op == "<=":
                return left <= right
            if op == ">":
                return left > right
            return left >= right
        raise BunRuntimeError("unsupported operands for '%s'" % op)
    if op == "==":
        return values_equal(left, right)
    if op == "!=":
        return not values_equal(left, right)
    raise BunRuntimeError("unknown operator '%s'" % op)


def unary_op(op, operand):
    if op == "-":
        if not is_number(operand):
            raise BunRuntimeError("unary '-' expects a number")
        return -operand
    if op == "!":
        return not is_truthy(operand)
    raise BunRuntimeError("unknown unary operator '%s'" % op)


def index_value(obj, index):
    if isinstance(obj, (list, str)):
        if not isinstance(index, int) or isinstance(index, bool):
            raise BunRuntimeError("index must be an integer")
        if index < 0 or index >= len(obj):
            raise BunRuntimeError("index %d out of range" % index)
        return obj[index]
    if isinstance(obj, BunDict):
        try:
            return obj.get(index)
        except KeyError:
            raise BunRuntimeError("key not found")
    raise BunRuntimeError("value is not indexable")


def assign_index(obj, index, value):
    if isinstance(obj, BunDict):
        obj.set(index, value)
        return
    if not isinstance(obj, list):
        raise BunRuntimeError("can only assign into a list")
    if not isinstance(index, int) or isinstance(index, bool):
        raise BunRuntimeError("list index must be an integer")
    if index < 0 or index >= len(obj):
        raise BunRuntimeError("list index %d out of range" % index)
    obj[index] = value


def type_name(value):
    if isinstance(value, bool):
        return "bool"
```

```python
        path, names = arg
        try:
            full_path = resolve_module_path(path, frame.base_dir)
            if full_path in self.modules:
                exports = self.modules[full_path]
            elif full_path in self.loading:
                raise BunRuntimeError("circular import of '%s'" % path)
            else:
                exports = self._run_module(path, full_path)
        except BunRuntimeError as error:
            if error.line == 0:
                error.line, error.col = line, col
            raise
        for name in names:
            if name not in exports:
                raise BunRuntimeError(
                    "module '%s' does not export '%s'" % (path, name))
            frame.env.define(name, exports[name], False)

    def _op_record_export(self, frame, arg, line, col):
        if not self.export_stack:
            raise BunRuntimeError("'export' is only allowed in modules")
        self.export_stack[-1][arg] = frame.env.get(arg)

    def _run_module(self, path, full_path):
        program = load_module_source(path, full_path)
        code = Compiler(full_path).compile_program(program)
        module_env = Environment(self.globals)
        module_base = module_dir(full_path)
        module_env.module_dir = module_base
        self.loading.add(full_path)
        self.export_stack.append({})
        try:
            try:
                self._execute(code, module_env, module_base)
            except BunRuntimeError as error:
                attach_module_frame(error, full_path, path)
                raise
        finally:
            exports = self.export_stack.pop()
            self.loading.discard(full_path)
        self.modules[full_path] = exports
        return exports
```

The unwinding logic in `_execute` (Task 1) truncates the stack to the handler's recorded depth, restores the recorded env, and pushes the message. The handler tuple carries a FOURTH element, the frame's iterator-stack depth at TRY time, and the unwind truncates it alongside the stack (`del target.iterators[iterator_depth:]`). This is a controller-mandated addition closing the deferred Task 3 review warning: without it, an error unwinding to a catch handler located outside a for-loop leaves a stale iterator on the frame, which can make an enclosing loop terminate early (e.g. `for i in [1, 2] { try { for j in [1] { throw "x" } } catch error { print("caught") } }` prints `caught` once instead of twice). Because iterator_depth is recorded per handler in the handler's own frame, the truncation can never remove a still-live outer loop's iterator (it sits at an index below the recorded depth).

```python
            if engine == "vm":
```
