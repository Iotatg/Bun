# Bun Phase 1.6 — Module System + First Repo Conversion

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Add Bun's module system (`import`/`export` with relative resolution and caching) and anonymous function expressions, then convert the first real file from the oven-sh/bun repository — `misctools/unicode-generator.ts` — into a Bun module, proving the language can express real tooling logic.

**Architecture:** Modules are Bun files executed once per absolute path into their own environment whose parent is the interpreter's globals; exported names (`export fn`/`export let`/`export var`) are collected into a per-module export table. `ImportStmt` resolves a relative path against the importing file's directory, loads (or reuses the cached) module, and binds the named imports immutably. Anonymous functions (`fn(x) { ... }` in expression position) create closures capturing their defining environment.

**Tech Stack:** Python 3.8+, pytest, standard library only.

## Global Constraints

- Python >=3.8; stdlib only; zero third-party deps.
- All runtime errors raise `BunRuntimeError` with line/col; no raw Python exceptions escape.
- Module paths must be relative (`./` or `../`); a `.bun` suffix is optional.
- Each module runs at most once per path (cache); circular imports raise `BunRuntimeError`.
- Imported bindings are immutable.
- No code comments unless the plan's code includes them.
- Use `python3` (no `python` on PATH).

    def _import_stmt(self):
        kw = self._advance()
        self._expect(TokenType.LBRACE, "expected '{' after 'import'")
        names = []
        if not self._check(TokenType.RBRACE):
            names.append(self._expect(TokenType.IDENT, "expected imported name").value)
            while self._match(TokenType.COMMA):
                names.append(
                    self._expect(TokenType.IDENT, "expected imported name").value)
        self._expect(TokenType.RBRACE, "expected '}' after import names")
        self._expect(TokenType.FROM, "expected 'from' after import list")
        path = self._expect(TokenType.STRING, "expected module path string")
        return ast.ImportStmt(names, path.value, kw.line, kw.col)

    def _exported_decl(self):
        self._advance()
        t = self._peek().type
        if t == TokenType.FN:
            decl = self._fn_decl()
        elif t in (TokenType.LET, TokenType.VAR):
            decl = self._var_decl(t == TokenType.VAR)
        else:
            tok = self._peek()
            raise ParseError(
                "'export' must precede 'fn', 'let', or 'var'", tok.line, tok.col)
        decl.exported = True
        return decl
```

- [ ] **Step 4: Write `bun/modules.py`**

```python
import os

from .errors import BunRuntimeError
from .lexer import Lexer
from .parser import Parser


def resolve_module_path(path, base_dir):
    if not (path.startswith("./") or path.startswith("../")):
        raise BunRuntimeError(
            "unsupported module path '%s' (only relative paths are allowed)" % path)
    full = os.path.normpath(os.path.join(base_dir, path))
    if not os.path.exists(full):
        full = full + ".bun"
    if not os.path.exists(full):
        raise BunRuntimeError("cannot find module '%s'" % path)
    return full


def load_module_source(path, full_path):
    try:
        with open(full_path, "r") as handle:
            source = handle.read()
    except OSError as error:
        raise BunRuntimeError(
            "cannot read module '%s': %s" % (path, error.strerror))
    return Parser(Lexer(source).tokenize()).parse()


def module_dir(full_path):
    return os.path.dirname(full_path)
```

        env = env if env is not None else self.globals
        if base_dir is None:
            base_dir = os.getcwd()
        self.dir_stack.append(base_dir)
        try:

            first = statements[0]
            raise BunRuntimeError(
                "'return' outside of a function",

        finally:
            self.dir_stack.pop()

```python
    def _record_export(self, name, value):
        if not self.export_stack:
            raise BunRuntimeError("'export' is only allowed in modules")
        self.export_stack[-1][name] = value

    def _load_module(self, path, line, col):
        base_dir = self.dir_stack[-1] if self.dir_stack else os.getcwd()
        try:
            full_path = modules.resolve_module_path(path, base_dir)
            program = modules.load_module_source(path, full_path)
        except BunRuntimeError as error:
            if error.line == 0:
                error.line, error.col = line, col
            raise
        if full_path in self.modules:
            return self.modules[full_path]
        if full_path in self.loading:
            raise BunRuntimeError("circular import of '%s'" % path)
        module_env = Environment(self.globals)
        self.loading.add(full_path)
        self.dir_stack.append(modules.module_dir(full_path))
        self.export_stack.append({})
        try:
            self.execute(program, module_env)
        finally:
            self.loading.discard(full_path)
            self.dir_stack.pop()
        exports = self.export_stack.pop()
        self.modules[full_path] = exports
        return exports
```

**Conversion mapping (from `misctools/unicode-generator.ts`):**
| TypeScript | Bun |
|---|---|
| `class Generator` with instance state | module with `generate()` taking all state locally |
| `Map<string, number> blockMap` | dict `block_map` keyed by joined block string |
| SHA-256 block hash | `block.join(",")` — exact content key, identical dedup result |
| `stage3.findIndex(...)` | `while` scan without early exit (no `break` in Bun) |
| `block.fill(0, blockLen)` | omitted: `block` is reset to zeros each finalize, so padding is always zero |
| `stage2.push(...block.slice(0, blockLen))` | push exactly `block_len` entries |
| template literals | string concatenation |
| `Generator.writeZig(...)` static | `export fn write_zig(...)` |

```bun
// Converted from bun-runtime/misctools/unicode-generator.ts
// (Generator class -> module functions; block dedup by SHA-256 -> exact
// block-content comparison, which yields identical tables.)

let BLOCK_SIZE = 256

export fn generate(ctx, max_codepoint) {
    var stage1 = []
    var stage2 = []
    var stage3 = []
    var block_map = {}

    var block = []
    var fill = 0
    while fill < BLOCK_SIZE {
        block.push(0)
        fill = fill + 1
    }

    var block_len = 0
    var cp = 0
    while cp <= max_codepoint {
        let elem = ctx["get"](cp)

        var block_idx = len(stage3)
        var scan = 0
        while scan < len(stage3) {
            if ctx["eql"](stage3[scan], elem) {
                block_idx = scan
            }
            scan = scan + 1
        }
        if block_idx == len(stage3) {
            stage3.push(elem)
        }

        block[block_len] = block_idx
        block_len = block_len + 1

        if block_len == BLOCK_SIZE or cp == max_codepoint {
            let key = block.join(",")
            if !block_map.has(key) {
                block_map[key] = len(stage2)
                var copy = 0
                while copy < block_len {
                    stage2.push(block[copy])
                    copy = copy + 1
                }
            }
            stage1.push(block_map[key])
            var reset = 0
            while reset < BLOCK_SIZE {
                block[reset] = 0
                reset = reset + 1
            }
            block_len = 0
        }
        cp = cp + 1
    }

    var tables = {}
    tables["stage1"] = stage1
    tables["stage2"] = stage2
    tables["stage3"] = stage3
    return tables
}

export fn write_zig(table_name, tables, elem_to_string) {
    var output = "/// Auto-generated. Do not edit.\n"
    output = output + "fn " + table_name + "(comptime Elem: type) type {\n"
    output = output + "    return struct {\n"
    output = output + "pub const stage1: [" + str(len(tables["stage1"]))
    output = output + "]u16 = .{"
    output = output + tables["stage1"].join(",")
    output = output + "};\n\n"
    output = output + "pub const stage2: [" + str(len(tables["stage2"]))
    output = output + "]u8 = .{"
    output = output + tables["stage2"].join(",")
    output = output + "};\n\n"
    output = output + "pub const stage3: [" + str(len(tables["stage3"]))
    output = output + "]Elem = .{"
    var rendered = []
    var i = 0
    while i < len(tables["stage3"]) {
        rendered.push(elem_to_string(tables["stage3"][i]))
        i = i + 1
    }
    output = output + rendered.join(",")
    output = output + "};\n"
    output = output + "    };\n}\n"
    return output
}
```

## Modules (Phase 1.6)

Any Bun file is a module. Mark declarations with `export` and import them by
relative path:

```bun
// greeting.bun
export fn hello(name) {
    return "Hello, " + name
}

export let punctuation = "!"

```bun
import { hello, punctuation } from "./greeting"

print(hello("Bun") + punctuation)

Rules:
- Module paths are relative to the importing file (`./` or `../`); the `.bun`

- Each module runs once per program; importers share its state.
- Imported bindings cannot be reassigned.
- Circular imports are an error.

Anonymous functions are expressions, so they can be stored in dicts or passed
around:

```bun
var ctx = {}
ctx["get"] = fn(x) {
    return x * 2
}
print(ctx["get"](21))
