# Bun Phase 4 Implementation Plan: Front-End Self-Hosting

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rewrite the Bun lexer, parser, and bytecode compiler in Bun itself, running the resulting plain-data code objects on the existing Python VM, with byte-identical parity against the Python front end across the whole corpus.

**Architecture:** A Bun driver (`selfhost/main.bun`) lexes, parses, and compiles a target program into the Phase 3.5 CodeObject format and hands it to the VM through a small Python bridge (`bun/selfhost.py`) exposing `args()`, `dump()`, and `run_code()`. The Python pipeline still lexes/parses/checks first so static errors remain unchanged and non-catchable. One canonical serializer lives only in the bridge, so both sides are compared through the same text format.

**Tech Stack:** Python >= 3.8 standard library only; Bun language (current feature set: no classes, no `break`/`continue`, no `isinstance`); pytest for tests.

## Global Constraints

- Python >= 3.8, standard library only; zero third-party runtime dependencies.
- Every language-level error is a `BunError` subclass; no raw Python exception may escape to the user.
- Static errors (`LexError`, `ParseError`, `BunTypeError`) are raised by the Python pipeline before the Bun front end runs and stay non-catchable.
- Data flies between the two sides as plain values: dicts, lists, strings, numbers, booleans, `null`. BunDictionaries are normalized by the bridge.
- The selfhost sources must obey the current Bun language: no classes, no `break`/`continue`, no direct dict iteration (use `.keys()`), no `isinstance` (dispatch on `kind`), no `isdigit`/`isalpha` (ASCII range checks), no `ord` (explicit escape table).
- No code comments in committed code.
- Execute on branch `bun-phase-4` (create it before Task 1).
- One commit per task; commit message style follows the repo (`feat:`, `fix:`, `chore:`, `docs:`, `test:`).
- After each task, run the full suite (`python3 -m pytest tests/ -q`) and record the pass count in the task's final step.
- The Python front end, the tree-walker, and the VM are not modified by this phase except for the `--front-end` flag in `bun/cli.py`.

## File Structure

- Create `selfhost/lexer.bun` — tokens as plain dicts; `tokenize(source)`.
- Create `selfhost/parser.bun` — tokens to AST dicts; `parse(tokens)`.
- Create `selfhost/compiler.bun` — AST dicts to CodeObject dicts; `compile_program(filename, source, program)`.
- Create `selfhost/main.bun` — driver: `args()` + `read_file` + mode dispatch (`lex`, `ast`, `dump`, `run`).
- Create `bun/selfhost.py` — bridge: `normalize`, `canonical`, `token_to_data`, `ast_to_data`, `code_to_data`, `run_front_end`, driver builtins.
- Modify `bun/cli.py` — `--front-end python|bun` flag, validation, and `run_file` branch (Task 4).
- Create `tests/test_selfhost.py` — canonical/normalize units, token parity, AST parity, code parity, self-compilation fixpoint, execution parity, error parity.
- Modify `tests/test_cli.py` — front-end flag validation and integration tests (Task 4).
- Docs: `README.md`, `docs/superpowers/specs/2026-10-14-bun-language-design.md` roadmap (Task 5).

```bun
let KEYWORDS = {
    "let": "LET",
    "var": "VAR",
    "fn": "FN",
    "if": "IF",
    "else": "ELSE",
    "while": "WHILE",
    "for": "FOR",
    "in": "IN",
    "return": "RETURN",
    "true": "TRUE",
    "false": "FALSE",
    "null": "NULL",
    "and": "AND",
    "or": "OR",
    "import": "IMPORT",
    "from": "FROM",
    "export": "EXPORT",
    "try": "TRY",
    "catch": "CATCH",
    "throw": "THROW",
}

let TWO_CHAR = {
    "==": "EQ",
    "!=": "NEQ",
    "<=": "LTE",
    ">=": "GTE",
    "->": "ARROW",
}

let ONE_CHAR = {
    "+": "PLUS",
    "-": "MINUS",
    "*": "STAR",
    "/": "SLASH",
    "%": "PERCENT",
    "=": "ASSIGN",
    "<": "LT",
    ">": "GT",
    "!": "BANG",
    "(": "LPAREN",
    ")": "RPAREN",
    "{": "LBRACE",
    "}": "RBRACE",
    "[": "LBRACKET",
    "]": "RBRACKET",
    ",": "COMMA",
    ":": "COLON",
    ".": "DOT",
}

let ESCAPES = {"n": "\n", "t": "\t", '"': '"', "\\": "\\"}

fn is_digit(ch) {
    return ch >= "0" and ch <= "9"
}

fn is_lower(ch) {
    return ch >= "a" and ch <= "z"
}

fn is_upper(ch) {
    return ch >= "A" and ch <= "Z"
}

fn is_alpha(ch) {
    return is_lower(ch) or is_upper(ch)
}

fn is_alnum(ch) {
    return is_alpha(ch) or is_digit(ch)
}

fn make_lexer(source) {
    return {"source": source, "pos": 0, "line": 1, "col": 1, "tokens": []}
}

fn peek_at(st, offset) {
    let index = st["pos"] + offset
    if index < st["source"].len() {
        return st["source"][index]
    }
    return ""
}

fn advance(st) {
    let ch = st["source"][st["pos"]]
    st["pos"] = st["pos"] + 1
    if ch == "\n" {
        st["line"] = st["line"] + 1
        st["col"] = 1
    } else {
        st["col"] = st["col"] + 1
    }
    return ch
}

fn add_token(st, token_type, value, line, col) {
    st["tokens"].push({"type": token_type, "value": value, "line": line, "col": col})
}

fn number(st) {
    let line = st["line"]
    let col = st["col"]
    let start = st["pos"]
    while is_digit(peek_at(st, 0)) {
        advance(st)
    }
    var is_float = false
    if peek_at(st, 0) == "." and is_digit(peek_at(st, 1)) {
        is_float = true
        advance(st)
        while is_digit(peek_at(st, 0)) {
            advance(st)
        }
    }
    let text = st["source"].slice(start, st["pos"])
    if is_float {
        add_token(st, "FLOAT", float(text), line, col)
    } else {
        add_token(st, "INT", int(text), line, col)
    }
}

fn string_token(st) {
    let line = st["line"]
    let col = st["col"]
    advance(st)
    let chars = []
    var done = false
    while !done {
        if st["pos"] >= st["source"].len() {
            throw "unterminated string at " + str(line) + ":" + str(col)
        }
        let ch = advance(st)
        if ch == '"' {
            done = true
        } else {
            if ch == "\\" {
                let esc_line = st["line"]
                let esc_col = st["col"]
                let nxt = advance(st)
                if !ESCAPES.has(nxt) {
                    throw "invalid escape at " + str(esc_line) + ":" + str(esc_col)
                }
                chars.push(ESCAPES.get(nxt))
            } else {
                chars.push(ch)
            }
        }
    }
    add_token(st, "STRING", chars.join(""), line, col)
}

fn identifier(st) {
    let line = st["line"]
    let col = st["col"]
    let start = st["pos"]
    while is_alnum(peek_at(st, 0)) or peek_at(st, 0) == "_" {
        advance(st)
    }
    let text = st["source"].slice(start, st["pos"])
    if KEYWORDS.has(text) {
        add_token(st, KEYWORDS.get(text), text, line, col)
    } else {
        add_token(st, "IDENT", text, line, col)
    }
}

fn operator(st) {
    let line = st["line"]
    let col = st["col"]
    let ch = advance(st)
    let two = ch + peek_at(st, 0)
    if TWO_CHAR.has(two) {
        advance(st)
        add_token(st, TWO_CHAR.get(two), two, line, col)
        return
    }
    if ONE_CHAR.has(ch) {
        add_token(st, ONE_CHAR.get(ch), ch, line, col)
        return
    }
    throw "unexpected character '" + ch + "' at " + str(line) + ":" + str(col)
}

export fn tokenize(source) {
    let st = make_lexer(source)
    while st["pos"] < st["source"].len() {
        let ch = peek_at(st, 0)
        if ch == " " or ch == "\t" or ch == "\r" or ch == "\n" {
            advance(st)
        } else {
            if ch == "#" or (ch == "/" and peek_at(st, 1) == "/") {
                while st["pos"] < st["source"].len() and peek_at(st, 0) != "\n" {
                    advance(st)
                }
            } else {
                if is_digit(ch) {
                    number(st)
                } else {
                    if ch == '"' {
                        string_token(st)
                    } else {
                        if is_alpha(ch) or ch == "_" {
                            identifier(st)
                        } else {
                            operator(st)
                        }
                    }
                }
            }
        }
    }
    add_token(st, "EOF", null, st["line"], st["col"])
    return st["tokens"]
}
```

```python
import os

from .checker import Checker
from .compiler import Compiler
from .dict import BunDict
from .errors import BunRuntimeError
from .lexer import Lexer
from .parser import Parser
from .values import Builtin
from .vm import VM

SELFHOST_DIR = os.path.join(
    os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "selfhost")

_DRIVER_CODE = None


def driver_code():
    global _DRIVER_CODE
    if _DRIVER_CODE is None:
        path = os.path.join(SELFHOST_DIR, "main.bun")
        with open(path, "r") as handle:
            source = handle.read()
        program = Parser(Lexer(source).tokenize()).parse()
        _DRIVER_CODE = Compiler(path, source).compile_program(program)
    return _DRIVER_CODE


def normalize(value):
    if isinstance(value, BunDict):
        return {
            normalize(key[1]): normalize(item)
            for key, item in value.items.items()
        }
    if isinstance(value, (list, tuple)):
        return [normalize(item) for item in value]
    return value


def canonical(value):
    value = normalize(value)
    if value is None:
        return "null"
    if value is True:
        return "true"
    if value is False:
        return "false"
    if isinstance(value, str):
        return _quote(value)
    if isinstance(value, float):
        return repr(value)
    if isinstance(value, int):
        return str(value)
    if isinstance(value, list):
        return "[" + ", ".join(canonical(item) for item in value) + "]"
    if isinstance(value, dict):
        items = sorted(value.items(), key=lambda pair: canonical(pair[0]))
        rendered = [canonical(key) + ": " + canonical(item) for key, item in items]
        return "{" + ", ".join(rendered) + "}"
    raise BunRuntimeError("cannot canonicalize %s" % type(value).__name__)


def _quote(text):
    escaped = text.replace("\\", "\\\\")
    escaped = escaped.replace('"', '\\"')
    escaped = escaped.replace("\n", "\\n")
    escaped = escaped.replace("\t", "\\t")
    escaped = escaped.replace("\r", "\\r")
    return '"' + escaped + '"'


def token_to_data(token):
    return {
        "type": token.type.name,
        "value": token.value,
        "line": token.line,
        "col": token.col,
    }


def _dump(args):
    if len(args) != 1:
        raise BunRuntimeError("dump() expects 1 argument")
    return canonical(args[0])


def run_front_end(path, mode="run", check=True):
    with open(path, "r") as handle:
        source = handle.read()
    if check:
        program = Parser(Lexer(source).tokenize()).parse()
        base_dir = os.path.dirname(os.path.abspath(path))
        Checker().check(program, base_dir=base_dir)
    vm = VM()
    vm.globals.define(
        "args", Builtin("args", lambda args: [path, mode]), False)
    vm.globals.define("dump", Builtin("dump", _dump), False)
    vm.run(driver_code(), base_dir=SELFHOST_DIR)
```

```bun
fn peek(st) {
    return st["tokens"][st["pos"]]
}

fn check(st, token_type) {
    return peek(st)["type"] == token_type
}

fn advance(st) {
    let tok = st["tokens"][st["pos"]]
    if tok["type"] != "EOF" {
        st["pos"] = st["pos"] + 1
    }
    return tok
}

fn in_types(items, value) {
    var i = 0
    while i < items.len() {
        if items[i] == value {
            return true
        }
        i = i + 1
    }
    return false
}

fn match_token(st, types) {
    if in_types(types, peek(st)["type"]) {
        return advance(st)
    }
    return null
}

fn expect(st, token_type, message) {
    if check(st, token_type) {
        return advance(st)
    }
    let tok = peek(st)
    throw message + ", found " + tok["type"]
}

fn statement(st) {
    let t = peek(st)["type"]
    if t == "IMPORT" {
        return import_stmt(st)
    }
    if t == "EXPORT" {
        return exported_decl(st)
    }
    if t == "LET" {
        return var_decl(st, false)
    }
    if t == "VAR" {
        return var_decl(st, true)
    }
    if t == "FN" {
        return fn_decl(st)
    }
    if t == "IF" {
        return if_stmt(st)
    }
    if t == "WHILE" {
        return while_stmt(st)
    }
    if t == "FOR" {
        return for_stmt(st)
    }
    if t == "RETURN" {
        return return_stmt(st)
    }
    if t == "TRY" {
        return try_stmt(st)
    }
    if t == "THROW" {
        return throw_stmt(st)
    }
    return expr_stmt(st)
}

fn var_decl(st, mutable) {
    let kw = advance(st)
    let name = expect(st, "IDENT", "expected variable name")
    let type_ann = opt_type(st)
    expect(st, "ASSIGN", "expected '=' after variable name")
    let value = expression(st)
    return {"kind": "LetStmt", "name": name["value"], "value": value, "mutable": mutable, "line": kw["line"], "col": kw["col"], "exported": false, "type_ann": type_ann}
}

fn fn_decl(st) {
    let kw = advance(st)
    let name = expect(st, "IDENT", "expected function name")
    expect(st, "LPAREN", "expected '(' after function name")
    let params = param_list(st)
    var return_type = null
    if match_token(st, ["ARROW"]) != null {
        return_type = type_name(st)
    }
    let body = block(st)
    return {"kind": "FnDecl", "name": name["value"], "params": params[0], "body": body, "line": kw["line"], "col": kw["col"], "exported": false, "param_types": params[1], "return_type": return_type}
}

fn param_list(st) {
    let names = []
    let types = []
    if !check(st, "RPAREN") {
        names.push(expect(st, "IDENT", "expected parameter name")["value"])
        types.push(opt_type(st))
        while match_token(st, ["COMMA"]) != null {
            names.push(expect(st, "IDENT", "expected parameter name")["value"])
            types.push(opt_type(st))
        }
    }
    expect(st, "RPAREN", "expected ')' after parameters")
    return [names, types]
}

fn opt_type(st) {
    if match_token(st, ["COLON"]) != null {
        return type_name(st)
    }
    return null
}

fn type_name(st) {
    if check(st, "FN") {
        return advance(st)["value"]
    }
    return expect(st, "IDENT", "expected type name")["value"]
}

fn import_stmt(st) {
    let kw = advance(st)
    expect(st, "LBRACE", "expected '{' after 'import'")
    let names = []
    if !check(st, "RBRACE") {
        names.push(expect(st, "IDENT", "expected imported name")["value"])
        while match_token(st, ["COMMA"]) != null {
            names.push(expect(st, "IDENT", "expected imported name")["value"])
        }
    }
    expect(st, "RBRACE", "expected '}' after import names")
    expect(st, "FROM", "expected 'from' after import list")
    let path = expect(st, "STRING", "expected module path string")
    return {"kind": "ImportStmt", "names": names, "path": path["value"], "line": kw["line"], "col": kw["col"]}
}

fn exported_decl(st) {
    advance(st)
    let t = peek(st)["type"]
    var decl = null
    if t == "FN" {
        decl = fn_decl(st)
    } else {
        if t == "LET" {
            decl = var_decl(st, false)
        } else {
            if t == "VAR" {
                decl = var_decl(st, true)
            } else {
                throw "'export' must precede 'fn', 'let', or 'var'"
            }
        }
    }
    decl["exported"] = true
    return decl
}

fn if_stmt(st) {
    let kw = advance(st)
    let condition = expression(st)
    let then_branch = block(st)
    var else_branch = null
    if match_token(st, ["ELSE"]) != null {
        if check(st, "IF") {
            else_branch = if_stmt(st)
        } else {
            else_branch = block(st)
        }
    }
    return {"kind": "IfStmt", "condition": condition, "then_branch": then_branch, "else_branch": else_branch, "line": kw["line"], "col": kw["col"]}
}

fn while_stmt(st) {
    let kw = advance(st)
    let condition = expression(st)
    let body = block(st)
    return {"kind": "WhileStmt", "condition": condition, "body": body, "line": kw["line"], "col": kw["col"]}
}

fn for_stmt(st) {
    let kw = advance(st)
    let name = expect(st, "IDENT", "expected loop variable name")
    expect(st, "IN", "expected 'in' after loop variable")
    let iterable = expression(st)
    let body = block(st)
    return {"kind": "ForStmt", "var_name": name["value"], "iterable": iterable, "body": body, "line": kw["line"], "col": kw["col"]}
}

fn return_stmt(st) {
    let kw = advance(st)
    var value = null
    if !check(st, "RBRACE") and !check(st, "EOF") and peek(st)["line"] == kw["line"] {
        value = expression(st)
    }
    return {"kind": "ReturnStmt", "value": value, "line": kw["line"], "col": kw["col"]}
}

fn try_stmt(st) {
    let kw = advance(st)
    let try_block = block(st)
    let params = []
    if match_token(st, ["CATCH"]) != null {
        if match_token(st, ["LPAREN"]) != null {
            let name = expect(st, "IDENT", "expected exception name")
            expect(st, "RPAREN", "expected ')' after exception name")
            params.push(name["value"])
        } else {
            if check(st, "IDENT") {
                params.push(advance(st)["value"])
            }
        }
        let catch_block = block(st)
        return {"kind": "TryStmt", "try_block": try_block, "catch_params": params, "catch_block": catch_block, "line": kw["line"], "col": kw["col"]}
    }
    throw "expected 'catch' after 'try' block"
}

fn throw_stmt(st) {
    let kw = advance(st)
    let value = expression(st)
    return {"kind": "ThrowStmt", "value": value, "line": kw["line"], "col": kw["col"]}
}

fn expr_stmt(st) {
    let start = peek(st)
    let expr = expression(st)
    if match_token(st, ["ASSIGN"]) != null {
        let value = expression(st)
        let kind = expr["kind"]
        if !(kind == "Identifier" or kind == "Index") {
            throw "invalid assignment target"
        }
        return {"kind": "AssignStmt", "target": expr, "value": value, "line": start["line"], "col": start["col"]}
    }
    return {"kind": "ExprStmt", "expr": expr, "line": start["line"], "col": start["col"]}
}

fn block(st) {
    let lbrace = expect(st, "LBRACE", "expected '{'")
    let statements = []
    while !check(st, "RBRACE") and !check(st, "EOF") {
        statements.push(statement(st))
    }
    expect(st, "RBRACE", "expected '}'")
    return {"kind": "Block", "statements": statements, "line": lbrace["line"], "col": lbrace["col"]}
}

fn expression(st) {
    return or_expr(st)
}

fn or_expr(st) {
    var expr = and_expr(st)
    while check(st, "OR") {
        let op = advance(st)
        expr = {"kind": "Logical", "left": expr, "op": "or", "right": and_expr(st), "line": op["line"], "col": op["col"]}
    }
    return expr
}

fn and_expr(st) {
    var expr = equality(st)
    while check(st, "AND") {
        let op = advance(st)
        expr = {"kind": "Logical", "left": expr, "op": "and", "right": equality(st), "line": op["line"], "col": op["col"]}
    }
    return expr
}

fn equality(st) {
    var expr = comparison(st)
    while in_types(["EQ", "NEQ"], peek(st)["type"]) {
        let op = advance(st)
        expr = {"kind": "Binary", "left": expr, "op": op["value"], "right": comparison(st), "line": op["line"], "col": op["col"]}
    }
    return expr
}

fn comparison(st) {
    var expr = term(st)
    while in_types(["LT", "LTE", "GT", "GTE"], peek(st)["type"]) {
        let op = advance(st)
        expr = {"kind": "Binary", "left": expr, "op": op["value"], "right": term(st), "line": op["line"], "col": op["col"]}
    }
    return expr
}

fn term(st) {
    var expr = factor(st)
    while in_types(["PLUS", "MINUS"], peek(st)["type"]) {
        let op = advance(st)
        expr = {"kind": "Binary", "left": expr, "op": op["value"], "right": factor(st), "line": op["line"], "col": op["col"]}
    }
    return expr
}

fn factor(st) {
    var expr = unary(st)
    while in_types(["STAR", "SLASH", "PERCENT"], peek(st)["type"]) {
        let op = advance(st)
        expr = {"kind": "Binary", "left": expr, "op": op["value"], "right": unary(st), "line": op["line"], "col": op["col"]}
    }
    return expr
}

fn unary(st) {
    if in_types(["MINUS", "BANG"], peek(st)["type"]) {
        let op = advance(st)
        return {"kind": "Unary", "op": op["value"], "operand": unary(st), "line": op["line"], "col": op["col"]}
    }
    return postfix(st)
}

fn postfix(st) {
    var expr = primary(st)
    var looping = true
    while looping {
        if check(st, "LPAREN") {
            advance(st)
            let args = []
            if !check(st, "RPAREN") {
                args.push(expression(st))
                while match_token(st, ["COMMA"]) != null {
                    args.push(expression(st))
                }
            }
            expect(st, "RPAREN", "expected ')' after arguments")
            expr = {"kind": "Call", "callee": expr, "args": args, "line": expr["line"], "col": expr["col"]}
        } else {
            if check(st, "LBRACKET") {
                advance(st)
                let index = expression(st)
                expect(st, "RBRACKET", "expected ']' after index")
                expr = {"kind": "Index", "target": expr, "index": index, "line": expr["line"], "col": expr["col"]}
            } else {
                if check(st, "DOT") {
                    advance(st)
                    let name = expect(st, "IDENT", "expected method name after '.'")
                    expect(st, "LPAREN", "expected '(' after method name")
                    let args = []
                    if !check(st, "RPAREN") {
                        args.push(expression(st))
                        while match_token(st, ["COMMA"]) != null {
                            args.push(expression(st))
                        }
                    }
                    expect(st, "RPAREN", "expected ')' after arguments")
                    expr = {"kind": "MethodCall", "receiver": expr, "method": name["value"], "args": args, "line": expr["line"], "col": expr["col"]}
                } else {
                    looping = false
                }
            }
        }
    }
    return expr
}

fn primary(st) {
    let tok = peek(st)
    let t = tok["type"]
    if t == "INT" or t == "FLOAT" or t == "STRING" {
        advance(st)
        return {"kind": "Literal", "value": tok["value"], "line": tok["line"], "col": tok["col"]}
    }
    if t == "TRUE" {
        advance(st)
        return {"kind": "Literal", "value": true, "line": tok["line"], "col": tok["col"]}
    }
    if t == "FALSE" {
        advance(st)
        return {"kind": "Literal", "value": false, "line": tok["line"], "col": tok["col"]}
    }
    if t == "NULL" {
        advance(st)
        return {"kind": "Literal", "value": null, "line": tok["line"], "col": tok["col"]}
    }
    if t == "IDENT" {
        advance(st)
        return {"kind": "Identifier", "name": tok["value"], "line": tok["line"], "col": tok["col"]}
    }
    if t == "FN" {
        advance(st)
        expect(st, "LPAREN", "expected '(' after 'fn'")
        let params = param_list(st)
        var return_type = null
        if match_token(st, ["ARROW"]) != null {
            return_type = type_name(st)
        }
        let body = block(st)
        return {"kind": "FnDecl", "name": "", "params": params[0], "body": body, "line": tok["line"], "col": tok["col"], "exported": false, "param_types": params[1], "return_type": return_type}
    }
    if t == "LPAREN" {
        advance(st)
        let expr = expression(st)
        expect(st, "RPAREN", "expected ')' after expression")
        return expr
    }
    if t == "LBRACE" {
        advance(st)
        let entries = []
        if !check(st, "RBRACE") {
            let key = expression(st)
            expect(st, "COLON", "expected ':' after dict key")
            let value = expression(st)
            entries.push([key, value])
            var more = true
            while more and match_token(st, ["COMMA"]) != null {
                if check(st, "RBRACE") {
                    more = false
                } else {
                    let next_key = expression(st)
                    expect(st, "COLON", "expected ':' after dict key")
                    let next_value = expression(st)
                    entries.push([next_key, next_value])
                }
            }
        }
        expect(st, "RBRACE", "expected '}' after dict entries")
        return {"kind": "DictLiteral", "entries": entries, "line": tok["line"], "col": tok["col"]}
    }
    if t == "LBRACKET" {
        advance(st)
        let elements = []
        if !check(st, "RBRACKET") {
            elements.push(expression(st))
            while match_token(st, ["COMMA"]) != null {
                elements.push(expression(st))
            }
        }
        expect(st, "RBRACKET", "expected ']' after list elements")
        return {"kind": "ListLiteral", "elements": elements, "line": tok["line"], "col": tok["col"]}
    }
    throw "unexpected token " + t
}

export fn parse(tokens) {
    let st = {"tokens": tokens, "pos": 0}
    let statements = []
    while !check(st, "EOF") {
        statements.push(statement(st))
    }
    return {"kind": "Program", "statements": statements}
}
```

```bun
fn make_code(name, params, code, filename, source) {
    return {"name": name, "params": params, "code": code, "filename": filename, "source": source}
}

fn emit(code, op, arg, node) {
    code.push([op, arg, node["line"], node["col"]])
    return code.len() - 1
}

fn patch(code, index) {
    let instruction = code[index]
    code[index] = [instruction[0], code.len(), instruction[2], instruction[3]]
}

export fn compile_program(filename, source, program) {
    let st = {"filename": filename, "source": source}
    let code = []
    let statements = program["statements"]
    var i = 0
    while i < statements.len() {
        compile_statement(st, statements[i], code)
        i = i + 1
    }
    code.push(["HALT", null, 0, 0])
    return make_code("<module>", [], code, filename, source)
}

fn compile_function(st, node) {
    let code = []
    emit(code, "ENTER_SCOPE", null, node)
    let body = node["body"]
    if body["kind"] == "Block" {
        let statements = body["statements"]
        var i = 0
        while i < statements.len() {
            compile_statement(st, statements[i], code)
            i = i + 1
        }
    } else {
        compile_statement(st, body, code)
    }
    emit(code, "EXIT_SCOPE", null, node)
    emit(code, "LOAD_CONST", null, node)
    emit(code, "RETURN", null, node)
    var name = node["name"]
    if name == "" {
        name = "anonymous"
    }
    return make_code(name, node["params"], code, st["filename"], st["source"])
}

fn compile_statement(st, node, code) {
    let kind = node["kind"]
    if kind == "LetStmt" {
        compile_expr(st, node["value"], code)
        emit(code, "DEFINE_NAME", [node["name"], node["mutable"]], node)
        if node["exported"] {
            emit(code, "RECORD_EXPORT", node["name"], node)
        }
        return
    }
    if kind == "AssignStmt" {
        let target = node["target"]
        if target["kind"] == "Identifier" {
            compile_expr(st, node["value"], code)
            emit(code, "STORE_NAME", target["name"], node)
        } else {
            if target["kind"] == "Index" {
                compile_expr(st, node["value"], code)
                compile_expr(st, target["target"], code)
                compile_expr(st, target["index"], code)
                emit(code, "STORE_INDEX", null, node)
            } else {
                throw "invalid assignment target"
            }
        }
        return
    }
    if kind == "ExprStmt" {
        compile_expr(st, node["expr"], code)
        emit(code, "POP", null, node)
        return
    }
    if kind == "Block" {
        emit(code, "ENTER_SCOPE", null, node)
        let statements = node["statements"]
        var i = 0
        while i < statements.len() {
            compile_statement(st, statements[i], code)
            i = i + 1
        }
        emit(code, "EXIT_SCOPE", null, node)
        return
    }
    if kind == "IfStmt" {
        compile_expr(st, node["condition"], code)
        let else_jump = emit(code, "JUMP_IF_FALSE", null, node)
        compile_branch(st, node["then_branch"], code)
        let end_jump = emit(code, "JUMP", null, node)
        patch(code, else_jump)
        if node["else_branch"] != null {
            compile_branch(st, node["else_branch"], code)
        }
        patch(code, end_jump)
        return
    }
    if kind == "WhileStmt" {
        let head = code.len()
        compile_expr(st, node["condition"], code)
        let exit_jump = emit(code, "JUMP_IF_FALSE", null, node)
        compile_branch(st, node["body"], code)
        emit(code, "JUMP", head, node)
        patch(code, exit_jump)
        return
    }
    if kind == "ForStmt" {
        compile_expr(st, node["iterable"], code)
        emit(code, "FOR_SETUP", null, node)
        let head = code.len()
        let exit_jump = emit(code, "FOR_NEXT", null, node)
        emit(code, "ENTER_SCOPE", null, node)
        emit(code, "DEFINE_NAME", [node["var_name"], true], node)
        compile_branch(st, node["body"], code)
        emit(code, "EXIT_SCOPE", null, node)
        emit(code, "JUMP", head, node)
        patch(code, exit_jump)
        return
    }
    if kind == "TryStmt" {
        let try_jump = emit(code, "TRY", null, node)
        emit(code, "ENTER_SCOPE", null, node)
        let try_statements = node["try_block"]["statements"]
        var i = 0
        while i < try_statements.len() {
            compile_statement(st, try_statements[i], code)
            i = i + 1
        }
        emit(code, "EXIT_SCOPE", null, node)
        emit(code, "END_TRY", null, node)
        let end_jump = emit(code, "JUMP", null, node)
        patch(code, try_jump)
        emit(code, "ENTER_SCOPE", null, node)
        let params = node["catch_params"]
        var p = 0
        while p < params.len() {
            emit(code, "DUP", null, node)
            emit(code, "DEFINE_NAME", [params[p], true], node)
            p = p + 1
        }
        emit(code, "POP", null, node)
        let catch_statements = node["catch_block"]["statements"]
        var c = 0
        while c < catch_statements.len() {
            compile_statement(st, catch_statements[c], code)
            c = c + 1
        }
        emit(code, "EXIT_SCOPE", null, node)
        patch(code, end_jump)
        return
    }
    if kind == "ThrowStmt" {
        compile_expr(st, node["value"], code)
        emit(code, "THROW", null, node)
        return
    }
    if kind == "FnDecl" {
        emit(code, "MAKE_FUNCTION", compile_function(st, node), node)
        emit(code, "DEFINE_NAME", [node["name"], false], node)
        if node["exported"] {
            emit(code, "RECORD_EXPORT", node["name"], node)
        }
        return
    }
    if kind == "ImportStmt" {
        emit(code, "IMPORT", [node["path"], node["names"]], node)
        return
    }
    if kind == "ReturnStmt" {
        if node["value"] != null {
            compile_expr(st, node["value"], code)
        } else {
            emit(code, "LOAD_CONST", null, node)
        }
        emit(code, "RETURN", null, node)
        return
    }
    throw "unknown statement: " + kind
}

fn compile_branch(st, node, code) {
    if node["kind"] == "Block" {
        emit(code, "ENTER_SCOPE", null, node)
        let statements = node["statements"]
        var i = 0
        while i < statements.len() {
            compile_statement(st, statements[i], code)
            i = i + 1
        }
        emit(code, "EXIT_SCOPE", null, node)
    } else {
        compile_statement(st, node, code)
    }
}

fn compile_expr(st, node, code) {
    let kind = node["kind"]
    if kind == "Literal" {
        emit(code, "LOAD_CONST", node["value"], node)
        return
    }
    if kind == "Identifier" {
        emit(code, "LOAD_NAME", node["name"], node)
        return
    }
    if kind == "ListLiteral" {
        let elements = node["elements"]
        var i = 0
        while i < elements.len() {
            compile_expr(st, elements[i], code)
            i = i + 1
        }
        emit(code, "BUILD_LIST", elements.len(), node)
        return
    }
    if kind == "Unary" {
        compile_expr(st, node["operand"], code)
        emit(code, "UNARY", node["op"], node)
        return
    }
    if kind == "Binary" {
        compile_expr(st, node["left"], code)
        compile_expr(st, node["right"], code)
        emit(code, "BINARY", node["op"], node)
        return
    }
    if kind == "Logical" {
        compile_expr(st, node["left"], code)
        emit(code, "DUP", null, node)
        var jump = 0
        if node["op"] == "and" {
            jump = emit(code, "JUMP_IF_FALSE", null, node)
        } else {
            jump = emit(code, "JUMP_IF_TRUE", null, node)
        }
        emit(code, "POP", null, node)
        compile_expr(st, node["right"], code)
        patch(code, jump)
        return
    }
    if kind == "Index" {
        compile_expr(st, node["target"], code)
        compile_expr(st, node["index"], code)
        emit(code, "INDEX", null, node)
        return
    }
    if kind == "MethodCall" {
        compile_expr(st, node["receiver"], code)
        let args = node["args"]
        var i = 0
        while i < args.len() {
            compile_expr(st, args[i], code)
            i = i + 1
        }
        emit(code, "METHOD_CALL", [node["method"], args.len()], node)
        return
    }
    if kind == "DictLiteral" {
        let entries = node["entries"]
        var i = 0
        while i < entries.len() {
            compile_expr(st, entries[i][0], code)
            compile_expr(st, entries[i][1], code)
            i = i + 1
        }
        emit(code, "BUILD_DICT", entries.len(), node)
        return
    }
    if kind == "Call" {
        compile_expr(st, node["callee"], code)
        let args = node["args"]
        var i = 0
        while i < args.len() {
            compile_expr(st, args[i], code)
            i = i + 1
        }
        emit(code, "CALL", args.len(), node)
        return
    }
    if kind == "FnDecl" {
        emit(code, "MAKE_FUNCTION", compile_function(st, node), node)
        return
    }
    throw "unknown expression: " + kind
}
```

```python
def _make_run_code(path):
    base_dir = os.path.dirname(os.path.abspath(path))

    def run_code(args):
        if len(args) != 1:
            raise BunRuntimeError("run_code() expects 1 argument")
        vm = VM()
        vm.run(normalize(args[0]), base_dir=base_dir)
        return None

    return Builtin("run_code", run_code)
```

```python
def run_file(path, check=True, engine="vm", front_end="python"):
    try:
        with open(path, "r") as handle:
            source = handle.read()
    except OSError as error:
        print("error: cannot open file '%s': %s" % (path, error), file=sys.stderr)
        return 1
    try:
        if front_end == "bun":
            selfhost.run_front_end(path, "run", check=check)
        else:
            run_source(source, filename=path, check=check, engine=engine)
        return 0
    except BunError as error:
        print(format_error(error, source, path), file=sys.stderr)
        return 1
```
